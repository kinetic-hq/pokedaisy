cmake_minimum_required(VERSION 3.18)
project(pokedaisy_linux C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Build only the pinned emulator library, never another platform frontend.
foreach(feature BUILD_QT BUILD_SDL BUILD_GL BUILD_GLES2 BUILD_GLES3 BUILD_SHARED USE_DISCORD_RPC USE_LUA USE_FFMPEG USE_LIBZIP USE_PNG USE_SQLITE3 USE_ELF USE_LZMA USE_DEBUGGERS)
  set(${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(BUILD_STATIC ON CACHE BOOL "" FORCE)
if(WIN32)
  # Windows (MinGW-w64): no zlib, so the portable zip carries SDL2.dll alone.
  set(USE_ZLIB OFF CACHE BOOL "" FORCE)
  # Library only: mGBA's frontends want epoxy on Windows, and the host is its own frontend.
  set(LIBMGBA_ONLY ON CACHE BOOL "" FORCE)
  set(M_CORE_GBA ON CACHE BOOL "" FORCE)
  set(M_CORE_GB ON CACHE BOOL "" FORCE)
endif()
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/mgba/CMakeLists.txt")
  message(FATAL_ERROR "Initialize the pinned core: git submodule update --init third_party/mgba")
endif()
add_subdirectory(../third_party/mgba mgba)
find_path(SDL2_INCLUDE_DIR SDL.h PATH_SUFFIXES SDL2 REQUIRED)
find_library(SDL2_LIBRARY NAMES SDL2 REQUIRED)
find_package(Threads REQUIRED)
file(GLOB PD_CORE CONFIGURE_DEPENDS ../3ds/core/*.c)
# RetroAchievements (ra.c): rcheevos as the Android app builds it, without its hashing (ra.c takes the file's MD5).
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/rcheevos/include/rc_client.h")
  message(FATAL_ERROR "Initialize rcheevos: git submodule update --init third_party/rcheevos")
endif()
file(GLOB RCHEEVOS_SOURCES CONFIGURE_DEPENDS ../third_party/rcheevos/src/*.c ../third_party/rcheevos/src/rapi/*.c
  ../third_party/rcheevos/src/rcheevos/*.c)
list(FILTER RCHEEVOS_SOURCES EXCLUDE REGEX "rc_libretro\\.c$|rc_client_external\\.c$|rc_client_raintegration\\.c$")
add_library(pd_rcheevos STATIC ${RCHEEVOS_SOURCES} ../third_party/rcheevos/src/rhash/md5.c)
target_include_directories(pd_rcheevos PUBLIC ../third_party/rcheevos/include)
target_compile_options(pd_rcheevos PRIVATE -w)
add_library(pd_linux_common STATIC bps.c ff_music.c shiny_odds.c host_io.c pocket_os.c wild.c fly.c field_items.c hud.c quick_menu.c view_layout.c updater.c http.c boxart.c save_edit.c game_grid.c game_support.c rom_list.c game_name.c ra.c screen_filter.c link_play.c shader.cpp ../third_party/xbrz/xbrz.cpp ${PD_CORE} ../3ds/desktop/pd_png.c ../app/src/main/cpp/pk_savevf.c
  ../app/src/main/cpp/pk_call.c ../app/src/main/cpp/pk_host.c ../app/src/main/cpp/pk_wide.c ../app/src/main/cpp/pk_wide_gb.c
  ../app/src/main/cpp/pk_rewind.c ../app/src/main/cpp/pk_link.c)
target_include_directories(pd_linux_common PUBLIC ../third_party/xbrz ../3ds/core ../third_party/mgba/include ${CMAKE_CURRENT_BINARY_DIR}/mgba/include ${SDL2_INCLUDE_DIR})
target_compile_definitions(pd_linux_common PUBLIC _GNU_SOURCE _POSIX_C_SOURCE=200809L)
if(WIN32)
  # libmgba uses Windows threads there (its threading.h picks them without USE_PTHREADS),
  # and the host's main is its own, not SDL2main's.
  # _WIN32_WINNT as libmgba's threading.h sets it, so including it after a system header is no redefinition.
  target_compile_definitions(pd_linux_common PUBLIC SDL_MAIN_HANDLED __USE_MINGW_ANSI_STDIO=1 _WIN32_WINNT=0x0600)
else()
  # plorpOS: libmgba is, and pk_rewind.c needs its threaded mCoreRewindContext
  target_compile_definitions(pd_linux_common PUBLIC USE_PTHREADS)
endif()
target_compile_options(pd_linux_common PRIVATE -Wall -Wextra)
# The app's EXPAND sources, built as they are.
# xBRZ as it comes (third_party/xbrz).
set_source_files_properties(../third_party/xbrz/xbrz.cpp PROPERTIES COMPILE_OPTIONS -w)
set_source_files_properties(../app/src/main/cpp/pk_wide.c ../app/src/main/cpp/pk_wide_gb.c PROPERTIES COMPILE_OPTIONS -Wno-unused-parameter)
target_link_libraries(pd_linux_common PUBLIC pd_rcheevos mgba ${SDL2_LIBRARY} m Threads::Threads ${CMAKE_DL_LIBS})
if(WIN32)
  target_link_libraries(pd_linux_common PUBLIC winhttp) # updater.c's release check and download
endif()
add_executable(pokedaisy-linux main.c)
# The release's version (desktop-release.yml passes the tag's); a local build says "dev".
set(POCKET_VERSION "dev" CACHE STRING "Version baked into the program")
target_compile_definitions(pokedaisy-linux PRIVATE POCKET_VERSION="${POCKET_VERSION}")
target_compile_options(pokedaisy-linux PRIVATE -Wall -Wextra -Werror)
target_link_libraries(pokedaisy-linux PRIVATE pd_linux_common)
if(WIN32)
  # pokedaisy.exe: a windowed program (its log goes to the console it was started
  # from, else DATA/pokedaisy.log), the UTF-8 code page and the icon from its
  # resource file; the C runtime linked in, so SDL2.dll is the only DLL beside it.
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  set(pd_ico ${CMAKE_CURRENT_BINARY_DIR}/pokedaisy.ico)
  add_custom_command(OUTPUT ${pd_ico}
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/windows/make_ico.py ${CMAKE_CURRENT_SOURCE_DIR}/icon.png ${pd_ico}
    DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/icon.png ${CMAKE_CURRENT_SOURCE_DIR}/windows/make_ico.py)
  configure_file(windows/pokedaisy.rc.in ${CMAKE_CURRENT_BINARY_DIR}/pokedaisy.rc @ONLY)
  target_sources(pokedaisy-linux PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/pokedaisy.rc ${pd_ico})
  set_target_properties(pokedaisy-linux PROPERTIES OUTPUT_NAME pokedaisy WIN32_EXECUTABLE ON)
  target_link_options(pokedaisy-linux PRIVATE -static)
endif()
include(CTest)
option(POCKET_DSPERATE "Optional DSperate Linux backend" OFF)
option(POCKET_MELONDS "Optional melonDS backend (Windows and Linux)" OFF)
if(POCKET_DSPERATE AND POCKET_MELONDS)
  message(FATAL_ERROR "Pick one DS backend: POCKET_DSPERATE or POCKET_MELONDS")
endif()
if(POCKET_DSPERATE OR POCKET_MELONDS)
  enable_language(CXX)
  set(CMAKE_CXX_STANDARD 17)
  include(dsperate/build.cmake)
  if(POCKET_MELONDS)
    set(ds_package_args --melonds)
  else()
    set(ds_package_args --dsperate)
  endif()
endif()
if(BUILD_TESTING)
  add_executable(linux-io-test test_host_io.c)
  add_executable(linux-runtime-test test_runtime.c)
  add_executable(linux-wild-test test_wild.c)
  add_executable(linux-view-test test_view_layout.c)
  add_executable(linux-updater-test test_updater.c)
  add_executable(linux-shader-test test_shader.c)
  add_executable(linux-rom-list-test test_rom_list.c)
  add_executable(linux-game-name-test test_game_name.c)
  add_executable(linux-boxart-folder-test test_boxart_folder.c)
  add_executable(linux-backdrop-test test_backdrop.c)
  add_executable(linux-wide-camera-test test_wide_camera.c)
  add_executable(linux-day-night-test test_day_night.c)
  add_executable(linux-bps-test test_bps.c)
  add_executable(linux-bezel-test test_bezel.c)
  add_executable(linux-moves-test test_moves.c)
  add_executable(linux-save-edit-test test_save_edit.c)
  add_executable(linux-rom-tweaks-test test_rom_tweaks.c)
  add_executable(linux-ff-music-test test_ff_music.c)
  add_executable(linux-shiny-odds-test test_shiny_odds.c)
  add_executable(linux-nuzlocke-test test_nuzlocke.c)
  add_executable(linux-screen-filter-test test_screen_filter.c)
  add_executable(linux-link-play-test test_link_play.c)
  foreach(test linux-io-test linux-runtime-test linux-wild-test linux-view-test linux-updater-test linux-shader-test linux-rom-list-test linux-game-name-test linux-boxart-folder-test linux-backdrop-test linux-wide-camera-test linux-day-night-test linux-bps-test linux-bezel-test linux-moves-test linux-save-edit-test linux-rom-tweaks-test linux-ff-music-test linux-shiny-odds-test linux-nuzlocke-test linux-screen-filter-test linux-link-play-test)
    target_compile_options(${test} PRIVATE -UNDEBUG -Wall -Wextra -Werror)
    target_link_libraries(${test} PRIVATE pd_linux_common)
    if(WIN32)
      target_link_options(${test} PRIVATE -static)
    endif()
    add_test(NAME ${test} COMMAND ${test})
  endforeach()
  if(NOT WIN32)
    target_link_libraries(linux-wide-camera-test PRIVATE m)
  endif()
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  if(WIN32)
    add_test(NAME package-smoke COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/windows/test_package.py $<TARGET_FILE:pokedaisy-linux> ${ds_package_args})
  else()
    add_test(NAME package-smoke COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_package.py $<TARGET_FILE:pokedaisy-linux> ${ds_package_args})
  endif()
  # The launcher starts the binary itself, which a cross build can't without its emulator's sysroot.
  if(NOT CMAKE_CROSSCOMPILING AND NOT WIN32)
    add_test(NAME portmaster-launcher COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_portmaster.py $<TARGET_FILE:pokedaisy-linux>)
  endif()
  add_test(NAME diagnostic-roms COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/make_diagnostic.py)
  # EDIT SAVE's move lists (save_tables_gen.h) still match the app's SaveTablesGen.kt.
  add_test(NAME save-tables-gen COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../scripts/gen_save_tables_c.py --check)
  set_tests_properties(diagnostic-roms PROPERTIES FIXTURES_SETUP diagnostics)
  set_tests_properties(linux-runtime-test PROPERTIES FIXTURES_REQUIRED diagnostics TIMEOUT 60)
  if(CMAKE_CROSSCOMPILING)
    set(headless offscreen)
  else()
    set(headless dummy)
  endif()
  set_tests_properties(linux-runtime-test PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=${headless};SDL_AUDIODRIVER=dummy;SDL_RENDER_DRIVER=software")
endif()
